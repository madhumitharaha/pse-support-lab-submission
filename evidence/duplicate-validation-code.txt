    67	func (v *Validator) Validate(ctx context.Context, seller Seller, importID string, g sheet.Group) []Error {
    68		var errs []Error
    69		for _, row := range g.Rows {
    70			errs = append(errs, v.validateRow(ctx, row)...)
    71		}
    72	
    73		first := g.Rows[0]
    74		if g.InvoiceNumber != "" {
    75			status, exists, err := v.invoices.ExistingStatus(ctx, seller.ID, g.InvoiceNumber, importID)
    76			if err != nil {
    77				v.log.Error("duplicate check failed", "invoice_number", g.InvoiceNumber, "error", err)
    78			} else if exists {
    79				errs = append(errs, Error{"AE-BR-09", sheet.ColInvoiceNumber,
    80					fmt.Sprintf("Document number '%s' already exists for this seller with status '%s'.", g.InvoiceNumber, status), first.Number})
    81			}
    82		}
