   146	func replay(key string) error {
   147		s, ok := findSample(key)
   148		if !ok {
   149			var keys []string
   150			for _, s := range dataset.Samples {
   151				keys = append(keys, s.Key)
   152			}
   153			return fmt.Errorf("unknown customer %q (choose one of: %s)", key, strings.Join(keys, ", "))
   154		}
   155		data, err := workbook.Build(s.Build("-" + runTag))
   156		if err != nil {
   157			return err
   158		}
   159		logf("%-12s uploading %s (%d KB)", key, s.Filename, len(data)/1024)
   160		imp, code, elapsed, err := upload(s.SellerID, s.Filename, data)
   161		if err != nil {
   162			logf("%-12s upload failed after %.1fs: %v", key, elapsed.Seconds(), err)
   163			return nil
   164		}
   165		logf("%-12s HTTP %d in %.1fs  import=%s  status=%s  invoices=%d", key, code, elapsed.Seconds(), imp.ID, imp.Status, imp.InvoiceCount)
   166		if final := settle(imp.ID, 90*time.Second); final != nil {
   167			logf("%-12s invoices: %s", key, formatCounts(final.StatusCounts))
   168		}
   169		return nil
   170	}
